# 🎯 标签（class）解释

该数据集中的样本属于意大利同一地区的三种不同的葡萄栽培者（cultivator）所生产的葡萄酒。每种酒的样本被标记为：

类别编号 类别名称 含义
0 class_0 第一种葡萄酒（59个样本）
1 class_1 第二种葡萄酒（71个样本）
2 class_2 第三种葡萄酒（48个样本）

🔍 注意：虽然具体 cultivator 的名字未公开披露，但每一类对应一个 cultivator（酒庄），即 class_0 表示第一个酒庄的葡萄酒，依此类推。

# 2. Dataloader

class WineDataLoader:  
    def __init__(self, batch_size):  
        self.i = 0  
        self.B = batch_size  
  
    def next_batch(self, x_data, y_data):  
        x = x_data[self.i: self.i + self.B]  
        y = y_data[self.i: self.i + self.B]  
        if self.i + self.B + 1 > x_data.size(0):  
            self.i = 0  
        return x, y

class WineModel(nn.Module):  
    def __init__(self):  
        super().__init__()  
        self.config = WineSelfAttentionModelConfig()  
        self.layers = nn.Sequential(OrderedDict({  
            '0linear': nn.Linear(13, self.config.n_hidden),  
            '1batchnorm': nn.BatchNorm1d(self.config.n_hidden),  
            '2relu':nn.ReLU(),  
            '3linear': nn.Linear(self.config.n_hidden, self.config.n_hidden),  
            '4batchnorm': nn.BatchNorm1d(self.config.n_hidden),  
            '5relu': nn.ReLU(),  
            '6decode': nn.Linear(self.config.n_hidden, 3)  
        })  
        )  
  
  
        self.global_step = {'count': 0}  

    def forward(self, x, y=None):  
        self.global_step['count'] += 1  
        logits = self.layers(x)  
        loss = None  
        if y is not None:  
            loss_fn = nn.CrossEntropyLoss()  
            loss = loss_fn(logits, y)  
        return logits, loss

def print_layer_shape_with_name(self, name):  
    def print_layer_shape(layer, args, output):  
        print(f"{name=}, {layer.__class__.__name__:20}: in: {str(args[0].shape):30}; out: {str(output.shape):30}")  
    return print_layer_shape  
  
def plot_layer_histogram_with_name(self, name):  
    os.makedirs('plot/weight', exist_ok=True)  
    os.makedirs('plot/input', exist_ok=True)  
    os.makedirs('plot/output', exist_ok=True)  
  
    def plot_layer_histogram(layer, args, output):  
        if hasattr(layer, 'weight'):  
            plt.figure()  
            hist, bin_edges = torch.histogram(layer.weight.view(-1), density=True)  
            plt.stairs(hist.detach().numpy(), bin_edges.detach().numpy(), fill=True)  
            plt.title(f"weight, step{self.global_step['count']}-{name=}")  
            plt.savefig(f"plot/weight/step{self.global_step['count']}-{name}")  
            plt.close()  
  
        plt.figure()  
        hist, bin_edges = torch.histogram(args[0].view(-1), density=True)  
        plt.stairs(hist.detach().numpy(), bin_edges.detach().numpy(), fill=True)  
        plt.title(  
            f"input, step: {self.global_step['count']} - {name=}")  
        plt.savefig(f"plot/input/step{self.global_step['count']}-{name}")  
        plt.close()  
  
        plt.figure()  
        hist, bin_edges = torch.histogram(output.view(-1), density=True)  
        plt.stairs(hist.detach().numpy(), bin_edges.detach().numpy(), fill=True)  
        plt.title(  
            f"output, step: {self.global_step['count']} - {name=}")  
        plt.savefig(f"plot/output/step{self.global_step['count']}-{name}.png")  
        plt.close()

# Write train.py

import matplotlib.pyplot as plt
import pandas as pd
import torch
from wine_model import WineModel, WineDataLoader
import evaluate

df = pd.read_csv("data.txt", header=None)

x_data = torch.tensor(df.iloc[:, 1:].values).float()
y_data = torch.tensor(df.iloc[:, 0].values) - 1

def train_test_split(x_data, y_data):
    idx = torch.randperm(x_data.size(0))
    x_data, y_data = x_data[idx, :], y_data[idx]
    cutoff = int(x_data.size(0) * 0.9)
    x_train, y_train = x_data[:cutoff], y_data[:cutoff]
    mean, std = x_train.mean(0), x_train.std(0)
    """
    (16, 10) => (1, 10)
    """
    x_train = (x_train - mean) / std
    x_test, y_test = x_data[cutoff:], y_data[cutoff:]
    x_test = (x_test - mean)/std

    return x_train, y_train, x_test, y_test

x_train, y_train, x_test, y_test = train_test_split(x_data, y_data)

def calc_metrics(y_pred, y_test):
    # print(f"{y_pred=}")
    # print(f"{y_test=}")
    accuracy_metrics = evaluate.load('accuracy')
    precision_metrics = evaluate.load('precision')
    recall_metrics = evaluate.load('recall')
    f1_metrics = evaluate.load('f1')

    accuracy = accuracy_metrics.compute(predictions=y_pred, references=y_test)
    precision = precision_metrics.compute(predictions=y_pred, references=y_test, average='macro')
    recall = recall_metrics.compute(predictions=y_pred, references=y_test, average='macro')
    f1 = f1_metrics.compute(predictions=y_pred, references=y_test, average='macro')
    return accuracy['accuracy'], precision['precision'], recall['recall'], f1['f1']



def train_model():
    model = WineModel()
    model.train()
    train_loss_lst = []
    eval_loss_lst = []
    optimizer = torch.optim.AdamW(params = model.parameters(),
                                  lr = 0.001)
    dataloader = WineDataLoader(batch_size=64)
    for i in range(1000):
        optimizer.zero_grad()
        x, y = dataloader.next_batch(x_train, y_train)
        # print(x, y)
        logits, loss = model(x, y)
        loss.backward()
        train_loss_lst.append(loss.item())
        optimizer.step()
        print(f"{loss.item() =}")
        # if i == 2:
        #     break
        if i % 100 == 99:
            eval_logits, loss  = model(x_test, y_test)
            print("eval loss:", loss)
            eval_loss_lst.append(loss.item())
            y_preds = eval_logits.argmax(dim=-1)
            metrics = calc_metrics(y_preds, y_test)
            print(f"{metrics=}")

    return model, train_loss_lst, eval_loss_lst


model, train_loss_lst, eval_loss_lst = train_model()
plt.plot(train_loss_lst)
plt.savefig('plot/train_loss_lst.png')
plt.close()
plt.plot(eval_loss_lst)
plt.savefig('plot/eval_loss_lst.png')
plt.close()
